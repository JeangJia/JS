JavaScript 对象解构 & 数组/对象嵌套解构

一、对象解构核心概念

对象解构是 ES6 语法糖,用于将对象的属性批量赋值给变量,核心是「按属性名匹配」(而非数组的「按位置匹配」)_ 变量名与对象属性名一致时,自动赋值;也可自定义变量名.

// 基础语法
let obj = { 属性1: 值1, 属性2: 值2 };
let { 属性1, 属性2 } = obj; // 按属性名匹配
let { 属性1: 自定义变量名, 属性2: 自定义变量名2 } = obj; // 自定义变量名

二、对象解构核心用法(含细节)

1. 基础解构:按属性名匹配赋值

变量名与对象属性名一致时,直接提取属性值;属性不存在则赋值为 undefined.

let obj = { name: '张三', age: 18, sex: '男' };
// 匹配同名属性
let { name, age, sex } = obj; // name='张三', age=18, sex='男'

// 属性不存在时
let { name, height } = obj; // height=undefined

2. 自定义变量名(重命名)

核心语法:{ 原属性名: 新变量名 },解决「变量名冲突」或「属性名不直观」问题(你的代码核心场景).

let obj = { name: '张三', age: 18 };
// 重命名:name → n,age → a
let { name: n, age: a } = obj; 
console.log(n); // 张三(不能再用 name 变量,已重命名)
console.log(a); // 18

// 你的代码示例
const { name: uname, age } = { name: 'pi', age: 6 };
console.log(uname); // pi(name 被重命名为 uname,直接用 name 会报错)
关键细节:重命名后,原属性名 不再作为变量存在,只能使用 新变量名 访问值.

3. 设置默认值

当对象属性不存在 / 为 undefined 时,使用默认值(与数组解构逻辑一致,但按属性名匹配).

let obj = { name: '张三', age: 18 };
// 属性存在,使用属性值
let { name: n2 = '李四', age: a2 = 20 } = obj; // n2='张三', a2=18

// 属性不存在,使用默认值
let { height: h = 180, weight: w = 70 } = obj; // h=180, w=70

// 属性为 undefined,使用默认值
let obj2 = { name: undefined, age: 18 };
let { name: n3 = '王五' } = obj2; // n3='王五'
细节:null/0/'' 等「假值」不会触发默认值,只有 undefined 或属性不存在才触发.

4. 解构已声明的变量

解构赋值给已存在的变量时,必须用 () 包裹解构语句(否则 JS 会将 {} 解析为代码块).

// 错误写法(语法错误)
let name, age;
{ name, age } = { name: '张三', age: 18 };

// 正确写法(加括号)
let name, age;
({ name, age } = { name: '张三', age: 18 }); // name='张三', age=18
注意:括号 () 必须紧贴赋值语句,否则仍会报错.

5. 剩余参数(...)解构剩余属性

用 ...变量名 接收对象剩余的所有属性,打包为一个新对象(ES2018 新增).

let obj = { name: '张三', age: 18, sex: '男', height: 175 };
// 提取 name,剩余属性打包为 restObj
let { name, ...restObj } = obj; 
console.log(name); // 张三
console.log(restObj); // { age: 18, sex: '男', height: 175 }
细节:对象剩余参数 ... 必须放在解构模式的最后一位,且只能有一个剩余参数.

6. 解构嵌套对象(单层)

对象属性值为对象时,解构模式需与嵌套结构一致(多层 {} 匹配多层对象).

let obj = {
    name: '张三',
    info: { age: 18, sex: '男' }
};
// 匹配嵌套结构:提取 info 中的 age 和 sex
let { info: { age, sex } } = obj; 
console.log(age, sex); // 18 男

// 同时提取外层和内层属性
let { name, info: { age: a } } = obj;
console.log(name, a); // 张三 18

三、数组 & 对象嵌套解构(核心重点)

实际开发中,数据常是「数组包对象」「对象包数组」「多层嵌套」,解构模式需严格匹配数据结构(数组用 [],对象用 {}).

场景1:对象中嵌套数组

数据结构:对象属性值为数组 → 解构时用 { 属性名: [变量1, 变量2] }.

// 数据:用户信息(爱好是数组)
let user = {
    name: '李四',
    age: 20,
    hobbies: ['篮球', '游戏', '读书']
};

// 解构:提取 name + hobbies 中的前两个爱好
let { name, hobbies: [h1, h2] } = user;
console.log(name); // 李四
console.log(h1, h2); // 篮球 游戏

// 进阶:提取 name + hobbies 中第二个爱好 + 剩余爱好
let { hobbies: [, h3, ...restHobbies] } = user;
console.log(h3); // 游戏
console.log(restHobbies); // ['读书']

场景2:数组中嵌套对象

数据结构:数组元素为对象 → 解构时用 [{ 属性1, 属性2 }, ...].

// 数据:商品列表(数组元素是商品对象)
let goods = [
    { id: 1, name: '手机', price: 2999 },
    { id: 2, name: '电脑', price: 5999 },
    { id: 3, name: '平板', price: 1999 }
];

// 解构:提取第一个商品的 name + 第二个商品的 price
let [{ name: g1Name }, { price: g2Price }] = goods;
console.log(g1Name); // 手机
console.log(g2Price); // 5999

// 进阶:提取第三个商品的 id + 重命名 price
let [, , { id: g3Id, price: g3Price }] = goods;
console.log(g3Id, g3Price); // 3 1999

场景3:多层嵌套(数组+对象混合)

数据结构:多层嵌套(如对象→数组→对象)→ 解构模式逐层匹配.

// 复杂数据:订单信息
let order = {
    orderId: 'OD20251128',
    user: {
        name: '王五',
        address: [
            { province: '广东省', city: '深圳市' },
            { province: '浙江省', city: '杭州市' }
        ]
    },
    goods: [{ name: '耳机', price: 199 }]
};

// 解构:提取 orderId + 第一个地址的 city + 商品的 name
let {
    orderId,
    user: {
        address: [{ city: addr1City }] // 数组第一个元素的 city
    },
    goods: [{ name: goodsName }] // 数组第一个元素的 name
} = order;

console.log(orderId); // OD20251128
console.log(addr1City); // 深圳市
console.log(goodsName); // 耳机

四、对象/嵌套解构常见坑与注意事项

  • 对象解构是「按名匹配」,与顺序无关:
    let { age, name } = { name: '张三', age: 18 } → 结果与 { name, age } 一致.
  • 解构 null/undefined 会报错:
    let { name } = null; / let { age } = undefined; → TypeError.
  • 嵌套解构时,外层属性必须存在:
    错误:let { info: { age } } = { name: '张三' } → info 不存在,报错「Cannot destructure property 'age' of 'undefined'」;
    解决:给外层属性加默认值 → let { info: { age } = {} } = { name: '张三' }(age=undefined,不报错).
  • 剩余参数不能用于数组/对象解构的非最后位置:
    错误:let { ...rest, name } = obj → SyntaxError.
  • 解构数字/字符串等原始值,会先转为对象:
    let { length } = 'hello'; → length=5(字符串的 length 属性);
    let { toString } = 123; → toString=Number.prototype.toString.

五、实战场景:解构接口返回的复杂数据

// 模拟接口返回的复杂数据
let res = {
    code: 200,
    data: {
        list: [
            { id: 1, title: '文章1', author: { name: '作者1', age: 30 } },
            { id: 2, title: '文章2', author: { name: '作者2', age: 25 } }
        ],
        total: 2
    },
    msg: 'success'
};

// 解构提取:total + 第一篇文章的 title + 第二篇文章作者的 name
let {
    data: {
        total,
        list: [
            { title: firstTitle }, // 第一篇文章的 title
            { author: { name: secondAuthorName } } // 第二篇文章作者的 name
        ]
    }
} = res;

console.log(total); // 2
console.log(firstTitle); // 文章1
console.log(secondAuthorName); // 作者2

六、核心总结

解构类型 匹配规则 核心语法 关键细节
对象解构 按属性名匹配 { a, b: c } = obj 重命名后原属性名不可用
数组解构 按位置匹配 [a, , b] = arr 空逗号跳过元素
嵌套解构 逐层匹配结构 { a: [b, { c }] } = obj 外层属性需存在(或加默认值)

核心原则:
1. 解构模式的「括号类型」([]/{})必须与数据结构一致;
2. 对象解构「按名」,数组解构「按位」;
3. 嵌套解构时,逐层拆解,避免访问不存在的层级.
